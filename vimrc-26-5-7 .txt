" ===================== 文件头自动插入 ==================
autocmd BufNewFile *.cpp,*.[ch],*.sh,*.java,*.go exec ":call SetTitle()"
 func SetTitle()
     if &filetype == 'sh'
         call setline(1,"#########################################################################")
         call append(line("."),   "# File Name:    ".expand("%"))
         call append(line(".")+1, "# Author:       程序员Ethan064")
         call append(line(".")+2, "# mail:         tli568227@gmail.com ")
         call append(line(".")+3, "# Created Time: ".strftime("%Y-%m-%d %H:%M:%S"))
         call append(line(".")+4, "#########################################################################")
         call append(line(".")+5, "#!/bin/bash")
         call append(line(".")+6, "")
     else
         call setline(1, "/* ************************************************************************")
         call append(line("."),   "* File Name:     ".expand("%"))
         call append(line(".")+1, "* Author:        程序员Ethan064")
         call append(line(".")+2, "* mail:          tli568227@gmail.com")
         call append(line(".")+3, "* Created Time:  ".strftime("%Y-%m-%d %H:%M:%S"))
         call append(line(".")+4, "* Description:   ")
         call append(line(".")+5, " ************************************************************************/")
         call append(line(".")+6, "")
     endif
 endfunc

" ================= Coc.nvim 配置 =================

" 1. 补全菜单外观与行为
" menu: 显示菜单, menuone: 只有一个选项也显示, noinsert: 不自动插入, noselect: 不自动选中
set completeopt=menu,menuone,noselect

" 2. 使用 Tab 键在补全菜单中导航 (可选，看个人习惯)
inoremap <silent><expr> <TAB>
      \ coc#pum#visible() ? coc#pum#next(1) :
      \ <SID>check_back_space() ? "\<TAB>" :
      \ coc#refresh()
inoremap <expr><S-TAB> coc#pum#visible() ? coc#pum#prev(1) : "\<S-TAB>"

function! s:check_back_space() abort
  let col = col('.') - 1
  return !col || getline('.')[col - 1]  =~# '\s'
endfunction

" 3. 使用 Enter 键确认补全
inoremap <silent><expr> <CR> coc#pum#visible() ? coc#pum#confirm() : "\<CR>"
